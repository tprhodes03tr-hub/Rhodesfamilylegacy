ROMULAN WARBIRD (ENEMY RAIDER - D'DERIDEX / BIRD-OF-PREY STYLE) --- function drawRomulanWarbird(x, y, angle) { ctx.save(); ctx.translate(x, y); ctx.rotate(angle); // Emerald Green Engine Glow ctx.shadowColor = '#00ff66'; ctx.shadowBlur = 8; // Split Double-Hull Wings (Emerald Green) ctx.fillStyle = '#1b4332'; ctx.strokeStyle = '#52b788'; ctx.lineWidth = 1.2; // Upper & Lower Curved Wing Arcs ctx.beginPath(); ctx.ellipse(-4, 0, 18, 14, 0, 0, Math.PI * 2); ctx.stroke(); // Hollow Core Cutout (Iconic D'Deridex Void) ctx.fillStyle = '#081c15'; ctx.beginPath(); ctx.ellipse(-4, 0, 10, 8, 0, 0, Math.PI * 2); ctx.fill(); // Lateral Nacelles ctx.fillStyle = '#00ff88'; ctx.fillRect(-14, -15, 14, 3); ctx.fillRect(-14, 12, 14, 3); // Forward Raptor Beak / Head Section ctx.fillStyle = '#2d6a4f'; ctx.beginPath(); ctx.moveTo(4, -6); ctx.lineTo(18, 0); ctx.lineTo(4, 6); ctx.lineTo(0, 0); ctx.closePath(); ctx.fill(); ctx.stroke(); // Disruptor Emitter Point ctx.fillStyle = '#74c69d'; ctx.beginPath(); ctx.arc(16, 0, 2, 0, Math.PI * 2); ctx.fill(); ctx.restore(); }
