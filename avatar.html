# MainGame.gd # Attach this script to a root Node3D in Godot 4. extends Node3D const DEFAULT_PORT: int = 8910 const DEFAULT_IP: str = "127.0.0.1" # Network & Game State var peer: ENetMultiplayerPeer var players_container: Node3D var projectiles_container: Node3D var ui_layer: CanvasLayer var status_label: Label func _ready() -> void: setup_environment() setup_networking_ui() players_container = Node3D.new() players_container.name = "Players" add_child(players_container) projectiles_container = Node3D.new() projectiles_container.name = "Projectiles" add_child(projectiles_container) multiplayer.peer_connected.connect(_on_peer_connected) multiplayer.peer_disconnected.connect(_on_peer_disconnected) # ============================================================================== # 1. PROCEDURAL 3D WORLD SETUP (Air, Water, & Lighting) # ============================================================================== func setup_environment() -> void: # Directional Sunlight var sun = DirectionalLight3D.new() sun.rotation_degrees = Vector3(-45, 30, 0) sun.light_energy = 1.2 add_child(sun) # Water Surface Visual Plane (Y = 0) var water_mesh = MeshInstance3D.new() var plane = PlaneMesh.new() plane.size = Vector2(800, 800) water_mesh.mesh = plane var water_mat = StandardMaterial3D.new() water_mat.albedo_color = Color(0.08, 0.42, 0.65, 0.65) water_mat.transparency = BaseMaterial3D.TRANSPARENCY_ALPHA water_mat.roughness = 0.15 water_mesh.material_override = water_mat water_mesh.position = Vector3(0, 0, 0) add_child(water_mesh) # ============================================================================== # 2. MULTIPLAYER LOBBY & UI # ============================================================================== func setup_networking_ui() -> void: ui_layer = CanvasLayer.new() add_child(ui_layer) var panel = PanelContainer.new() panel.position = Vector2(20, 20) ui_layer.add_child(panel) var vbox = VBoxContainer.new() panel.add_child(vbox) status_label = Label.new() status_label.text = "Avatar Combat: Host or Join" vbox.add_child(status_label) var host_btn = Button.new() host_btn.text = "Host Session (Server)" host_btn.pressed.connect(_on_host_pressed) vbox.add_child(host_btn) var ip_input = LineEdit.new() ip_input.text = DEFAULT_IP ip_input.placeholder_text = "Server IP" vbox.add_child(ip_input) var join_btn = Button.new() join_btn.text = "Join Session (Client)" join_btn.pressed.connect(func(): _on_join_pressed(ip_input.text)) vbox.add_child(join_btn) func _on_host_pressed() -> void: peer = ENetMultiplayerPeer.new() var err = peer.create_server(DEFAULT_PORT, 8) if err != OK: status_label.text = "Error hosting: " + str(err) return multiplayer.multiplayer_peer = peer status_label.text = "Hosting on port " + str(DEFAULT_PORT) spawn_player_avatar(1) func _on_join_pressed(ip_address: String) -> void: peer = ENetMultiplayerPeer.new() var address = ip_address if ip_address != "" else DEFAULT_IP var err = peer.create_client(address, DEFAULT_PORT) if err != OK: status_label.text = "Error joining: " + str(err) return multiplayer.multiplayer_peer = peer status_label.text = "Connecting to " + address + "..." func _on_peer_connected(id: int) -> void: if multiplayer.is_server(): # Spawn existing players for the newcomer and spawn the newcomer spawn_player_avatar(id) for child in players_container.get_children(): rpc_id(id, "sync_spawn_player", child.name.to_int(), child.global_position) func _on_peer_disconnected(id: int) -> void: if players_container.has_node(str(id)): players_container.get_node(str(id)).queue_free() func spawn_player_avatar(id: int, start_pos: Vector3 = Vector3(0, 15, 0)) -> void: if multiplayer.is_server(): rpc("sync_spawn_player", id, start_pos) @rpc("call_local", "reliable") func sync_spawn_player(id: int, start_pos: Vector3) -> void: if players_container.has_node(str(id)): return var player = AvatarWarrior.new() player.name = str(id) player.global_position = start_pos player.authority_id = id player.main_ref = self players_container.add_child(player) # ============================================================================== # 3. COMBAT PROJECTILE LOGIC (High-Speed Archery) # ============================================================================== @rpc("any_peer", "call_local", "reliable") func spawn_network_arrow(shooter_id: int, spawn_pos: Vector3, launch_dir: Vector3) -> void: var arrow = ArrowProjectile.new() arrow.global_position = spawn_pos arrow.direction = launch_dir.normalized() arrow.shooter_id = shooter_id arrow.main_ref = self projectiles_container.add_child(arrow) @rpc("any_peer", "call_local", "reliable") func execute_aoe_melee_strike(origin: Vector3, radius: float, damage: float, attacker_id: int) -> void: for player in players_container.get_children(): if player is AvatarWarrior and player.authority_id != attacker_id: if player.global_position.distance_to(origin) <= radius: player.take_damage(damage) # ============================================================================== # 4. AVATAR WARRIOR CLASS (Flight Archery & Water Breach) # ============================================================================== class AvatarWarrior extends CharacterBody3D: enum Stance { AIR_FLIGHT, AQUATIC_DIVE } var authority_id: int = 1 var main_ref: Node3D var current_stance: Stance = Stance.AIR_FLIGHT # Movement & Combat parameters var flight_speed: float = 38.0 var water_speed: float = 24.0 var breach_velocity_boost: float = 32.0 var health: float = 100.0 var camera: Camera3D var mesh_inst: MeshInstance3D var target_sync_pos: Vector3 func _ready() -> void: target_sync_pos = global_position # Avatar Visual Representation mesh_inst = MeshInstance3D.new() var capsule = CapsuleMesh.new() capsule.radius = 0.7 capsule.height = 2.4 mesh_inst.mesh = capsule var mat = StandardMaterial3D.new() # Blue Na'vi tint mat.albedo_color = Color(0.15, 0.45, 0.85) if authority_id == 1 else Color(0.12, 0.65, 0.75) mesh_inst.material_override = mat add_child(mesh_inst) var col = CollisionShape3D.new() var col_shape = CapsuleShape3D.new() col_shape.radius = 0.7 col_shape.height = 2.4 col.shape = col_shape add_child(col) # Setup local player camera if authority_id == multiplayer.get_unique_id(): camera = Camera3D.new() camera.position = Vector3(0, 3.5, 7.5) camera.rotation_degrees = Vector3(-18, 0, 0) add_child(camera) camera.current = true func _physics_process(delta: float) -> void: if authority_id != multiplayer.get_unique_id(): global_position = global_position.lerp(target_sync_pos, 0.25) return handle_mode_transitions() match current_stance: Stance.AIR_FLIGHT: handle_aerial_flight(delta) Stance.AQUATIC_DIVE: handle_aquatic_swim(delta) move_and_slide() rpc("sync_position_to_peers", global_position, rotation) func handle_mode_transitions() -> void: # Auto-switch to diving when beneath water level if global_position.y < 0.0 and current_stance == Stance.AIR_FLIGHT: current_stance = Stance.AQUATIC_DIVE # Aquatic Breach Launch: Surface breach when pressing Jump/Space near Y=0 if current_stance == Stance.AQUATIC_DIVE and global_position.y >= -1.0: if Input.is_action_pressed("ui_accept"): current_stance = Stance.AIR_FLIGHT velocity.y = breach_velocity_boost velocity.z *= 1.5 func handle_aerial_flight(_delta: float) -> void: var forward = -transform.basis.z var right = transform.basis.x var move_input = Vector2.ZERO if Input.is_key_pressed(KEY_W): move_input.y += 1.0 if Input.is_key_pressed(KEY_S): move_input.y -= 1.0 if Input.is_key_pressed(KEY_D): move_input.x += 1.0 if Input.is_key_pressed(KEY_A): move_input.x -= 1.0 # Pitch and Yaw controls if Input.is_key_pressed(KEY_UP): rotate_x(deg_to_rad(1.5)) if Input.is_key_pressed(KEY_DOWN): rotate_x(deg_to_rad(-1.5)) if Input.is_key_pressed(KEY_LEFT): rotate_y(deg_to_rad(2.0)) if Input.is_key_pressed(KEY_RIGHT): rotate_y(deg_to_rad(-2.0)) # Scene 1: Precision Focus Aim (Slow-Mo) if Input.is_mouse_button_pressed(MOUSE_BUTTON_RIGHT): Engine.time_scale = 0.4 else: Engine.time_scale = 1.0 # Aerial Shooting if Input.is_action_just_pressed("ui_select") or Input.is_mouse_button_pressed(MOUSE_BUTTON_LEFT): main_ref.rpc("spawn_network_arrow", authority_id, global_position + forward * 2.0, forward) velocity = forward * (flight_speed * (1.0 + move_input.y * 0.4)) + right * (move_input.x * 12.0) func handle_aquatic_swim(_delta: float) -> void: var forward = -transform.basis.z var swim_dir = Vector3.ZERO if Input.is_key_pressed(KEY_W): swim_dir += forward if Input.is_key_pressed(KEY_S): swim_dir -= forward if Input.is_key_pressed(KEY_E): swim_dir.y += 1.0 # Swim up if Input.is_key_pressed(KEY_Q): swim_dir.y -= 1.0 # Dive down velocity = swim_dir.normalized() * water_speed # Scene 2: Heavy Aquatic/Breach Melee Strike if Input.is_key_pressed(KEY_F): main_ref.rpc("execute_aoe_melee_strike", global_position, 5.5, 45.0, authority_id) @rpc("unreliable") func sync_position_to_peers(pos: Vector3, rot: Vector3) -> void: target_sync_pos = pos rotation = rot func take_damage(amount: float) -> void: health -= amount if health <= 0: global_position = Vector3(0, 20, 0) # Respawn health = 100.0 # ============================================================================== # 5. PROJECTILE ENTITY # ============================================================================== class ArrowProjectile extends Node3D: var speed: float = 110.0 var direction: Vector3 = Vector3.FORWARD var shooter_id: int = 1 var main_ref: Node3D var lifetime: float = 3.5 func _ready() -> void: var mesh = MeshInstance3D.new() var box = BoxMesh.new() box.size = Vector3(0.08, 0.08, 1.6) mesh.mesh = box var mat = StandardMaterial3D.new() mat.albedo_color = Color(1.0, 0.7, 0.2) mat.emission_enabled = true mat.emission = Color(1.0, 0.5, 0.1) mat.emission_energy_multiplier = 3.0 mesh.material_override = mat add_child(mesh) func _process(delta: float) -> void: global_position += direction * speed * delta lifetime -= delta # Check collisions against peers if main_ref and multiplayer.is_server(): for player in main_ref.players_container.get_children(): if player is AvatarWarrior and player.authority_id != shooter_id: if global_position.distance_to(player.global_position) < 1.8: player.take_damage(35.0) queue_free() return if lifetime <= 0.0: queue_free()
